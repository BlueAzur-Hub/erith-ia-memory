/* Snapshot Analyse — canonical graph series in read-only memory.
   No fetch, no DOM data scraping, no second graph, no orders. */
function atlasPublishedChartSeriesRead(){
 try{
  const engine=state.chartEngineV2;
  const drawing=engine?.realChart;
  if(!drawing||els.mainChart?.closest?.(".chart-shell")?.dataset?.chartState!=="valid")
   return null;
  const external=atlasExternalChartContext.active===true;
  let coin,period,result,origin,revision;
  if(external){
   const x=atlasExternalChartContext;
   coin=x.coin;period=Number(x.period);result=x.result;
   if(!coin?.id||x.loading||x.error||x.resultPeriod!==period||
      !result||engine.loading||engine.lastRenderedKey!=="external:"+coin.id+":"+period)
    return null;
   origin="external";revision=x.token;
  }else{
   const broker=state.dataBroker?.chart;
   coin=getSelectedCoin();period=Number(broker?.period);result=broker?.result;
   if(!coin?.id||state.graphSelectionCleared||atlasComparisonActive()||
      engine.loading||broker?.status!=="ready"||broker.error||
      !atlasChartContextMatches(broker)||broker.coinId!==coin.id||
      Number(state.chartPeriodDays)!==period||
      ![atlasChartKey(coin,period),broker.contextKey].includes(engine.lastRenderedKey))
    return null;
   origin="market";revision=state.chartRenderToken;
  }
  if(result?.blocked||result?.integrity?.ok===false||
     !Array.isArray(result?.series)||result.series.length<2)return null;
  const rows=atlasSanitizeChartRows(result.series);
  const plotted=drawing.data?.datasets?.find(d=>d.type==="line"&&
    String(d.atlasCoin?.id||"")===String(coin.id));
  if(rows.length<2||!plotted||!Array.isArray(plotted.data)||
     plotted.data.length!==rows.length)return null;
  // Compare series to canonical Chart.js source metadata, not the pixels,
  // base-100 projection or a virtual live price.
  for(let i=0;i<rows.length;i++)
   if(rows[i][0]!==Number(plotted.data[i]?.x)||
      rows[i][1]!==Number(plotted.data[i]?.rawPrice))return null;
  const currency=String(result.currency||result.quoteCurrency||"").toUpperCase()||null;
  const quote=String(coin.rawQuoteCurrency||"").toUpperCase()||null;
  const conversionRequired=!!quote&&!!currency&&quote!==currency;
  const points=Object.freeze(rows.map(r=>Object.freeze([r[0],r[1]])));
  return Object.freeze({
   assetId:String(coin.id),symbol:String(coin.symbol||"").toUpperCase(),
   instrument:String(coin.providerContext?.providerSymbol||coin.listingPair||"").toUpperCase()||null,
   provider:String(result.provider||result.sourceFamily||"")||null,
   source:String(result.source||"")||null,
   sourceMode:String(result.sourceMode||"")||null,
   periodDays:period,currency,quoteCurrency:quote,
   conversion:Object.freeze({required:conversionRequired,
    verified:!conversionRequired&&!!currency,
    source:conversionRequired?null:result.source||null,
    observedAt:conversionRequired?null:result.generatedAt||null}),
   published:true,integrityOk:true,origin,
   firstPointAt:new Date(points[0][0]).toISOString(),
   lastPointAt:new Date(points[points.length-1][0]).toISOString(),
   selectionRevision:revision,chartRevision:revision,series:points
  });
 }catch(_){return null;}
}
globalThis.AgentCryptoPublishedSeries=Object.freeze({
 read:atlasPublishedChartSeriesRead,
 read_only:true,fetches:0,storage_writes:0,real_orders:false
});
